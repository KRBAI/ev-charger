# Outbound: Listen for remote commands
def on_cmd_snapshot(docs, changes, read_time):
    for change in changes:
        if change.type.name in ["ADDED", "MODIFIED"]:
            data = change.document.to_dict()
            if data.get("status") == "PENDING":
                print(f"Executing: {data.get('command')} on {data.get('charger_id')}")
                change.document.reference.update({"status": "EXECUTED"})

db.collection("Commands").where("status", "==", "PENDING").on_snapshot(on_cmd_snapshot)
